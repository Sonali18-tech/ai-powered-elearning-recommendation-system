
FROM node:22-alpine

WORKDIR /app

COPY package*.json ./
RUN npm ci

COPY . .

# Prisma CLI reads DATABASE_URL through prisma.config.ts.
# This placeholder is only for generating Prisma Client at build time.
ENV DATABASE_URL="postgresql://user:password@localhost:5432/learnai"

RUN npx prisma generate

ENV NODE_ENV=production
ENV PORT=5000

EXPOSE 5000

CMD ["npm", "start"]
